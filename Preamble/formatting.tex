% HEADER & FOOTER (Fancyhdr)
% ====================================================
\setlength{\headheight}{14.5pt}  % ADD THIS LINE to fix the warning
\addtolength{\topmargin}{-2.5pt} % ADD THIS LINE to compensate

\fancyhf{} % clear

\fancypagestyle{mydefault}{%
    \fancyhf{} % clear

    % Right header: Chapter Title (uppercase disabled)
    \fancyhead[R]{\textbf{\nouppercase{\leftmark}}}
    \renewcommand{\headrulewidth}{0.4pt}
    
    % Right footer: Title
    \fancyfoot[L]{\textbf{\nouppercase{\rightmark}}}
    
    % Right footer: page number
    \fancyfoot[R]{\thepage}
    \renewcommand{\footrulewidth}{0.4pt}

}

% Apply the style globally
\pagestyle{mydefault}

% Format of chapter marks
\renewcommand{\chaptermark}[1]{%
    \markboth{Chapter \thechapter: #1}{#1}
}

% Remove section marks
\renewcommand{\sectionmark}[1]{}

% Ensure plain pages (chapter starting pages, etc.) also use our style
\fancypagestyle{plain}{\pagestyle{mydefault}}
